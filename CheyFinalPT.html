Project: Math Quiz Game

import java.util.Random;
import java.util.Scanner;

public class MathQuiz {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        Random rand = new Random();

        int score = 0;
        int totalQuestions = 5;

        System.out.println("=== RANDOM MATH QUIZ GAME ===");
        System.out.println("Answer with A, B, C, or D.\n");

        for (int q = 1; q <= totalQuestions; q++) {
            // Random numbers and random operation
            int a = rand.nextInt(12) + 1;
            int b = rand.nextInt(12) + 1;
            int op = rand.nextInt(3);
            int answer;
            char sign;

            if (op == 0) {
                answer = a + b;
                sign = '+';
            } else if (op == 1) {
                if (a < b) {
                    int temp = a;
                    a = b;
                    b = temp;
                }
                answer = a - b;
                sign = '-';
            } else {
                answer = a * b;
                sign = 'x';
            }

            int[] choices = new int[4];
            int correctIndex = rand.nextInt(4);

            for (int i = 0; i < 4; i++) {
                if (i == correctIndex) {
                    choices[i] = answer;
                } else {
                    int wrong;
                    boolean valid;
                    do {
                        wrong = answer + rand.nextInt(11) - 5;
                        valid = (wrong != answer);
                        for (int j = 0; j < i; j++) {
                            if (choices[j] == wrong) {
                                valid = false;
                            }
                        }
                    } while (!valid);
                    choices[i] = wrong;
                }
            }

            System.out.println("Question " + q + ": What is " + a + " " + sign + " " + b + "?");
            System.out.println("A. " + choices[0]);
            System.out.println("B. " + choices[1]);
            System.out.println("C. " + choices[2]);
            System.out.println("D. " + choices[3]);

            String userAnswer;
            do {
                System.out.print("Your answer: ");
                userAnswer = input.next().toUpperCase();
            } while (!userAnswer.equals("A") && !userAnswer.equals("B")
                    && !userAnswer.equals("C") && !userAnswer.equals("D"));

            String correctLetter = "" + (char) ('A' + correctIndex);

            if (userAnswer.equals(correctLetter)) {
                System.out.println("Correct!\n");
                score++;
            } else {
                System.out.println("Wrong! The correct answer is " + correctLetter + ". " + answer + "\n");
            }
        }

        System.out.println("=== GAME OVER ===");
        System.out.println("Your score: " + score + " out of " + totalQuestions);

        input.close();
    }
}
